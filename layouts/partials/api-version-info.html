{{/* Where a page sits in a project's versioned API reference. Returns a dict: inapi, id, name (the version this page documents, "" when not in the API), url_root (the root of this page's tree), rel (the page's path below that root), proj (the project's `api` data from assemble.py), latest, indexed (only the default /generated/api/ copy goes to search). */}} {{ $path := strings.TrimSuffix "/" .Path }} {{ $id := .Section }} {{ $res := dict "inapi" false "indexed" true }} {{ $proj := dict }} {{ with index site.Data.projects $id }}{{ with .api }}{{ $proj = . }}{{ end }}{{ end }} {{ $prefix := printf "/%s/generated/api" $id }} {{ if and $id (or (eq $path $prefix) (strings.HasPrefix $path (printf "%s/" $prefix))) }} {{ $rest := strings.TrimPrefix $prefix $path }} {{ $name := "" }} {{ $root := printf "%s/" $prefix }} {{ $m := findRE `^/(dev|v[0-9][^/]*)(/|$)` $rest 1 }} {{ if $m }} {{ $name = index (findRE `^/[^/]+` $rest 1) 0 | strings.TrimPrefix "/" }} {{ $root = printf "%s/%s/" $prefix $name }} {{ $rest = strings.TrimPrefix (printf "/%s" $name) $rest }} {{ end }} {{ $res = merge $res (dict "inapi" true "id" $id "name" $name "url_root" $root "rel" (strings.TrimPrefix "/" $rest) "proj" $proj "latest" ($proj.latest | default "") "indexed" (eq $name "")) }} {{ end }} {{ return $res }}
